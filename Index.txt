  <script>
    /**
     * STATO GLOBALE DELL'APPLICAZIONE
     */
    const state = {
      // Posizione dell'utente
      userLat: null,
      userLon: null,
      userAlt: 500, // Fallback altitudine in metri se il GPS non fornisce altitude
      hasAltitude: false,

      // Orientamento dispositivo in gradi
      azimuth: 0,         // Direzione bussola (0° = Nord, 90° = Est, 180° = Sud, 270° = Ovest)
      pitch: 0,           // Inclinazione verticale (0° = telefono verticale puntato all'orizzonte, + su, - giù)
      compassOffset: 0,   // Correzione manuale utente (-30° / +30°)

      // Parametri ottici stimati fotocamera
      horizontalFOV: 60,  // Campo visivo orizzontale standard smartphone (~60°)

      // Database vette caricate
      peaks: [],
      isSimulatedData: false,
      overpassLoaded: false
    };

    /**
     * RIFERIMENTI AGLI ELEMENTI DOM
     */
    const dom = {
      startScreen: document.getElementById('start-screen'),
      startErrorBox: document.getElementById('start-error-box'),
      btnStart: document.getElementById('btn-start'),
      video: document.getElementById('camera-feed'),
      touchLayer: document.getElementById('touch-layer'),
      horizonLine: document.getElementById('horizon-line'),
      tapMarker: document.getElementById('tap-marker'),
      tapAngleBadge: document.getElementById('tap-angle-badge'),
      gpsDot: document.getElementById('gps-dot'),
      gpsStatusText: document.getElementById('gps-status-text'),
      peaksCountBadge: document.getElementById('peaks-count-badge'),
      hudAzimuth: document.getElementById('hud-azimuth'),
      hudPitch: document.getElementById('hud-pitch'),
      hudAltitude: document.getElementById('hud-altitude'),
      btnToggleCalib: document.getElementById('btn-toggle-calib'),
      calibrationDrawer: document.getElementById('calibration-drawer'),
      compassOffsetSlider: document.getElementById('compass-offset-slider'),
      calibValueLabel: document.getElementById('calib-value-label'),
      btnResetCalib: document.getElementById('btn-reset-calib'),
      hintPill: document.getElementById('hint-pill'),
      resultSheet: document.getElementById('result-sheet'),
      btnCloseSheet: document.getElementById('btn-close-sheet'),
      matchBadge: document.getElementById('match-badge'),
      matchAccuracy: document.getElementById('match-accuracy'),
      peakName: document.getElementById('peak-name'),
      peakStatsGrid: document.getElementById('peak-stats-grid'),
      peakElevation: document.getElementById('peak-elevation'),
      peakDistance: document.getElementById('peak-distance'),
      peakBearing: document.getElementById('peak-bearing'),
      alternativesContainer: document.getElementById('alternatives-container'),
      alternativesList: document.getElementById('alternatives-list'),
      toastBanner: document.getElementById('toast-banner')
    };

    /**
     * 1. AVVIO APPLICAZIONE E RICHIESTA PERMESSI
     * Gestisce fotocamera, sensori di movimento (incluso iOS 13+ requestPermission) e GPS.
     */
    dom.btnStart.addEventListener('click', async () => {
      dom.startErrorBox.classList.add('hidden');
      dom.btnStart.disabled = true;
      dom.btnStart.innerHTML = '<span>Inizializzazione sensori...</span>';

      try {
        // A. Richiesta permessi orientamento su iOS Safari (deve avvenire direttamente dentro un evento click)
        if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') {
          const permissionState = await DeviceOrientationEvent.requestPermission();
          if (permissionState !== 'granted') {
            throw new Error('Permesso bussola/giroscopio negato. Abilita i sensori di movimento nelle impostazioni di Safari.');
          }
        }

        // B. Avvio Fotocamera Posteriore
        await startCamera();

        // C. Avvio Listener Sensori (Bussola & Giroscopio)
        startOrientationTracking();

        // D. Avvio Geolocalizzazione GPS
        startGeolocation();

        // Nascondi schermata iniziale
        dom.startScreen.classList.add('hidden');
      } catch (err) {
        console.error('Errore avvio PeakTap AR:', err);
        dom.startErrorBox.textContent = err.message || 'Impossibile accedere alla fotocamera o ai sensori. Verifica di usare HTTPS.';
        dom.startErrorBox.classList.remove('hidden');
        dom.btnStart.disabled = false;
        dom.btnStart.innerHTML = '<span>Riprova Avvio</span>';
      }
    });

    /**
     * Attiva lo stream video della fotocamera posteriore ('environment')
     */
    async function startCamera() {
      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        throw new Error('Il tuo browser non supporta l\'accesso alla fotocamera (getUserMedia).');
      }

      const constraints = {
        audio: false,
        video: {
          facingMode: { ideal: 'environment' },
          width: { ideal: 1920 },
          height: { ideal: 1080 }
        }
      };

      const stream = await navigator.mediaDevices.getUserMedia(constraints);
      dom.video.srcObject = stream;

      // Prova a leggere il campo visivo reale se esposto dalle capabilities del track video
      try {
        const videoTrack = stream.getVideoTracks()[0];
        if (videoTrack && typeof videoTrack.getCapabilities === 'function') {
          const caps = videoTrack.getCapabilities();
          if (caps.fieldOfView) {
            state.horizontalFOV = caps.fieldOfView;
          }
        }
      } catch (e) {
        // Mantiene il valore predefinito di 60°
      }
    }

    /**
     * 2. LETTURA BUSSOLA (AZIMUT) E INCLINAZIONE (PITCH)
     */
    function startOrientationTracking() {
      // Su Android Chrome 'deviceorientationabsolute' fornisce il Nord geografico reale
      window.addEventListener('deviceorientationabsolute', handleOrientation, true);
      // Fallback standard e iOS Safari (che espone webkitCompassHeading)
      window.addEventListener('deviceorientation', handleOrientation, true);
    }

    function handleOrientation(event) {
      let heading = null;

      // 1. iOS Safari fornisce direttamente la bussola corretta rispetto al Nord magnetico
      if (typeof event.webkitCompassHeading === 'number') {
        heading = event.webkitCompassHeading;
      } 
      // 2. Android / Standard W3C: alpha va in senso antiorario, quindi 360 - alpha
      else if (event.alpha !== null) {
        // Se l'evento non è assoluto e abbiamo già ricevuto eventi assoluti, ignoriamo
        if (event.type === 'deviceorientation' && event.absolute === false) {
          return;
        }
        heading = (360 - event.alpha) % 360;
      }

      if (heading !== null) {
        // Applica eventuale offset di calibrazione manuale dell'utente
        const calibratedAzimuth = (heading + state.compassOffset + 360) % 360;
        // Smoothing leggero (filtro passa-basso) per evitare tremolii
        state.azimuth = smoothAngle(state.azimuth, calibratedAzimuth, 0.25);
      }

      // Calcolo dell'inclinazione verticale (Pitch) quando il telefono è tenuto in verticale (Portrait)
      // event.beta = 90° quando il telefono è perfettamente verticale (puntato all'orizzonte, 0° elevazione)
      // event.beta > 90° quando si inclina verso l'alto (elevazione positiva)
      // event.beta < 90° quando si inclina verso il basso (elevazione negativa)
      if (typeof event.beta === 'number') {
        const screenAngle = (screen.orientation && screen.orientation.angle) || window.orientation || 0;
        let rawPitch = 0;

        if (screenAngle === 0) {
          rawPitch = event.beta - 90;
        } else if (screenAngle === 90) {
          rawPitch = -event.gamma - 90;
        } else if (screenAngle === -90 || screenAngle === 270) {
          rawPitch = event.gamma - 90;
        } else {
          rawPitch = event.beta - 90;
        }

        // Limita pitch tra -85° e +85° per evitare singolarità allo zenit/nadir
        rawPitch = Math.max(-85, Math.min(85, rawPitch));
        state.pitch = state.pitch * 0.75 + rawPitch * 0.25;
      }

      updateHUD();
    }

    /**
     * Interpolazione angolare fluida che gestisce correttamente il passaggio 359° <-> 0°
     */
    function smoothAngle(current, target, factor) {
      let diff = ((target - current + 540) % 360) - 180;
      return (current + diff * factor + 360) % 360;
    }

    /**
     * Aggiorna i valori telemetrici nell'HUD superiore e sposta la linea dell'orizzonte
     */
    function updateHUD() {
      const azRounded = Math.round(state.azimuth);
      const cardinal = getCardinalDirection(azRounded);
      dom.hudAzimuth.textContent = `${azRounded}° ${cardinal}`;

      const pitchSign = state.pitch > 0 ? '+' : '';
      dom.hudPitch.textContent = `${pitchSign}${state.pitch.toFixed(1)}°`;

      // Aggiorna posizione visiva della linea dell'orizzonte sullo schermo
      const verticalFOV = getVerticalFOV();
      const offsetPercent = (state.pitch / verticalFOV) * 100;
      const horizonTop = Math.max(5, Math.min(95, 50 + offsetPercent));
      dom.horizonLine.style.top = `${horizonTop}%`;
    }

    /**
     * 3. GEOLOCALIZZAZIONE GPS E SCARICAMENTO VETTE (OVERPASS API)
     */
    function startGeolocation() {
      if (!('geolocation' in navigator)) {
        activateFallbackLocation('GPS non supportato dal browser');
        return;
      }

      navigator.geolocation.watchPosition(
        async (position) => {
          const { latitude, longitude, altitude, accuracy } = position.coords;
          const isFirstFix = state.userLat === null;

          state.userLat = latitude;
          state.userLon = longitude;

          if (typeof altitude === 'number' && !isNaN(altitude)) {
            state.userAlt = Math.round(altitude);
            state.hasAltitude = true;
            dom.hudAltitude.textContent = `${state.userAlt} m`;
          } else {
            dom.hudAltitude.textContent = `~${state.userAlt} m`;
          }

          dom.gpsDot.className = 'w-2.5 h-2.5 rounded-full bg-emerald-400 shrink-0';
          dom.gpsStatusText.textContent = `GPS attivo (±${Math.round(accuracy)}m)`;

          // Scarica le vette una volta ottenuto il primo fix GPS valido
          if (isFirstFix && !state.overpassLoaded) {
            state.overpassLoaded = true;
            await fetchNearbyPeaks(latitude, longitude);
          }
        },
        (error) => {
          console.warn('Errore GPS:', error);
          if (state.userLat === null) {
            activateFallbackLocation('Segnale GPS assente (Modalità Demo)');
          }
        },
        {
          enableHighAccuracy: true,
          maximumAge: 10000,
          timeout: 15000
        }
      );
    }

    /**
     * Se il GPS non è disponibile (es. test indoor su desktop), imposta una posizione demo sulle Alpi
     */
    async function activateFallbackLocation(reason) {
      // Coordinate demo: vista spettacolare sul gruppo del Monte Bianco / Courmayeur
      state.userLat = 45.7969;
      state.userLon = 6.9689;
      state.userAlt = 1224;
      dom.gpsDot.className = 'w-2.5 h-2.5 rounded-full bg-amber-400 shrink-0';
      dom.gpsStatusText.textContent = reason;
      dom.hudAltitude.textContent = `${state.userAlt} m`;
      showToast('Posizione simulata (Courmayeur) attivata per test.');
      await fetchNearbyPeaks(state.userLat, state.userLon);
    }

    /**
     * Interroga Overpass API di OpenStreetMap per ottenere tutte le vette (natural=peak) entro 50 km
     */
    async function fetchNearbyPeaks(lat, lon) {
      dom.peaksCountBadge.textContent = 'Caricamento vette...';
      const radiusMeters = 50000; // 50 km

      const query = `
        [out:json][timeout:15];
        node["natural"="peak"]["name"](around:${radiusMeters},${lat},${lon});
        out body;
      `;

      try {
        const response = await fetch('https://overpass-api.de/api/interpreter', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/x-www-form-urlencoded;charset=UTF-8'
          },
          body: 'data=' + encodeURIComponent(query)
        });

        if (!response.ok) {
          throw new Error(`HTTP error ${response.status}`);
        }

        const data = await response.json();
        const parsedPeaks = (data.elements || [])
          .filter(el => el.tags && el.tags.name)
          .map(el => {
            // Estrae altitudine (può essere "3841", "3841 m", ecc.)
            let ele = parseFloat(String(el.tags.ele || '').replace(',', '.').replace(/[^\d.-]/g, ''));
            if (isNaN(ele)) ele = null;

            return {
              id: el.id,
              name: el.tags.name,
              lat: el.lat,
              lon: el.lon,
              ele: ele
            };
          });

        if (parsedPeaks.length > 0) {
          state.peaks = parsedPeaks;
          state.isSimulatedData = false;
          dom.peaksCountBadge.textContent = `${state.peaks.length} vette OSM`;
          showToast(`Caricate ${state.peaks.length} vette reali entro 50 km.`);
        } else {
          // Nessuna montagna trovata nel raggio di 50 km (es. pianura o mare): genera vette simulate
          loadSimulatedPeaks(lat, lon);
        }
      } catch (err) {
        console.warn('Overpass API non raggiungibile, uso vette simulate di fallback:', err);
        loadSimulatedPeaks(lat, lon);
      }
    }

    /**

     * l'app anche in città di pianura o offline.

    function loadSimulatedPeaks(lat, lon) {
      const mockTemplates = [

        { name: 'Gran Cresta (Demo)', bearing: 252, distKm: 38, ele: 3290 },

    /**
     * 4. LOGICA DI RICONOSCIMENTO AL TOCCO SULLO SCHERMO (RAYCASTING ANGOLARE)
     */
    dom.touchLayer.addEventListener('click', (event) => {
      if (state.userLat === null || state.peaks.length === 0) {
        showToast('Attendi il caricamento del GPS e delle vette...');
        return;
      }

      const rect = dom.touchLayer.getBoundingClientRect();
      const clickX = event.clientX - rect.left;
      const clickY = event.clientY - rect.top;

      // Calcola l'offset normalizzato del tocco rispetto al centro dello schermo [-0.5, +0.5]
      const normX = (clickX / rect.width) - 0.5;
      const normY = 0.5 - (clickY / rect.height); // + verso l'alto, - verso il basso

      // Calcola gli angoli FOV orizzontale e verticale
      const hFOV = state.horizontalFOV;
      const vFOV = getVerticalFOV();

      // Angoli esatti corrispondenti al pixel toccato sullo schermo
      const tapAzimuth = (state.azimuth + normX * hFOV + 360) % 360;
      const tapElevation = state.pitch + normY * vFOV;

      // Posiziona e anima il mirino visivo sul punto toccato
      showTapMarker(clickX, clickY, tapAzimuth, tapElevation);

      // Cerca le vette più vicine a (tapAzimuth, tapElevation)
      identifyPeakAtAngles(tapAzimuth, tapElevation);
    });

    /**
     * Calcola il campo visivo verticale in base all'aspect ratio del viewport
     */
    function getVerticalFOV() {
      const w = window.innerWidth || 1;
      const h = window.innerHeight || 1;
      return state.horizontalFOV * (h / w);
    }

    /**
     * Confronta gli angoli del tocco con tutte le vette nel database
     */
    function identifyPeakAtAngles(tapAzimuth, tapElevation) {
      const MAX_TOLERANCE_DEG = 8.5; // Raggio di tolleranza in gradi attorno al dito

      const candidates = state.peaks.map(peak => {
        const distKm = haversineDistanceKm(state.userLat, state.userLon, peak.lat, peak.lon);
        const bearing = calculateBearing(state.userLat, state.userLon, peak.lat, peak.lon);
        const elevationAngle = calculateElevationAngle(state.userAlt, peak.ele, distKm);

        // Differenza angolare orizzontale (gestendo il wrap-around a 360°)
        let dAz = ((bearing - tapAzimuth + 540) % 360) - 180;
        // Differenza angolare verticale
        let dEl = elevationAngle - tapElevation;

        // Distanza angolare euclidea totale in gradi
        const angularDistance = Math.sqrt(dAz * dAz + dEl * dEl);

        return {
          ...peak,
          distKm,
          bearing,
          elevationAngle,
          angularDistance
        };
      })
      .filter(p => p.angularDistance <= MAX_TOLERANCE_DEG)
      .sort((a, b) => a.angularDistance - b.angularDistance);

      renderIdentificationResult(candidates, tapAzimuth, tapElevation);
    }

    /**
     * Aggiorna e apre la scheda inferiore (Bottom Sheet) con i dettagli della vetta trovata

    function renderIdentificationResult(candidates, tapAz, tapEl) {
      dom.hintPill.classList.add('hidden');

      if (candidates.length === 0) {
        // Nessuna vetta trovata nel cono del tocco

        dom.peakName.textContent = 'Nessuna cima catalogata qui';
        dom.peakStatsGrid.classList.add('hidden');
        dom.alternativesContainer.classList.add('hidden');
        return;
      }

      // Mostra la vetta migliore (distanza angolare minima)
      const best = candidates[0];
      dom.peakStatsGrid.classList.remove('hidden');

      dom.matchBadge.textContent = state.isSimulatedData ? 'Vetta Demo Identificata' : 'Vetta Identificata';
      dom.matchBadge.className = 'px-2.5 py-0.5 rounded-full text-[11px] font-bold uppercase tracking-wide bg-emerald-500/20 text-emerald-300 border border-emerald-400/30';
      dom.matchAccuracy.textContent = `Scarto angolare: ${best.angularDistance.toFixed(1)}°`;

      displayPeakDetails(best);

      // Se ci sono altre vette vicine al punto toccato (es. cresta affollata), mostrale come alternative cliccabili
      if (candidates.length > 1) {
        dom.alternativesContainer.classList.remove('hidden');
        dom.alternativesList.innerHTML = '';

        candidates.slice(1, 5).forEach(alt => {
          const btn = document.createElement('button');

            dom.matchAccuracy.textContent = `Scarto angolare: ${alt.angularDistance.toFixed(1)}°`;
            displayPeakDetails(alt);
          });
          dom.alternativesList.appendChild(btn);
        });
      } else {
        dom.alternativesContainer.classList.add('hidden');
      }
    }

    /**
     * 5. FUNZIONI MATEMATICHE GEODETICHE
     */

    // Distanza in km tra due coordinate GPS (Formula di Haversine)
    function haversineDistanceKm(lat1, lon1, lat2, lon2) {
      const R = 6371.0; // Raggio medio terrestre in km
      const dLat = toRad(lat2 - lat1);
      const dLon = toRad(lon2 - lon1);
      const a =
        Math.sin(dLat / 2) * Math.sin(dLat / 2) +
        Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
        Math.sin(dLon / 2) * Math.sin(dLon / 2);
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
      return R * c;
    }

    // Azimut iniziale (bearing) da punto A a punto B in gradi [0..360]
    function calculateBearing(lat1, lon1, lat2, lon2) {
      const phi1 = toRad(lat1);
      const phi2 = toRad(lat2);
      const dLambda = toRad(lon2 - lon1);

      const y = Math.sin(dLambda) * Math.cos(phi2);
      const x = Math.cos(phi1) * Math.sin(phi2) -
                Math.sin(phi1) * Math.cos(phi2) * Math.cos(dLambda);
      const theta = Math.atan2(y, x);
      return (toDeg(theta) + 360) % 360;
    }

    // Angolo di elevazione apparente tenendo conto della differenza di quota e della curvatura terrestre
    function calculateElevationAngle(userAltMeters, peakAltMeters, distKm) {
      const distMeters = Math.max(10, distKm * 1000);
      const targetAlt = peakAltMeters !== null ? peakAltMeters : (userAltMeters + 1000);

      // Correzione curvatura terrestre + rifrazione atmosferica standard (k ≈ 0.13)
      // Abbassamento apparente h_curv = d^2 / (2 * R_eff), con R_eff ≈ 6371000 / (1 - 0.13) ≈ 7323000 m
      const earthCurvatureDrop = (distMeters * distMeters) / (2 * 7323000);
      const deltaAlt = (targetAlt - userAltMeters) - earthCurvatureDrop;

      return toDeg(Math.atan2(deltaAlt, distMeters));
    }

    // Calcola una coordinata di destinazione data partenza, distanza (km) e azimut (gradi)
    function destinationPoint(lat, lon, distKm, bearingDeg) {
      const R = 6371.0;
      const delta = distKm / R;
      const theta = toRad(bearingDeg);
      const phi1 = toRad(lat);
      const lambda1 = toRad(lon);

      const phi2 = Math.asin(
        Math.sin(phi1) * Math.cos(delta) +
        Math.cos(phi1) * Math.sin(delta) * Math.cos(theta)
      );
      const lambda2 = lambda1 + Math.atan2(
        Math.sin(theta) * Math.sin(delta) * Math.cos(phi1),
        Math.cos(delta) - Math.sin(phi1) * Math.sin(phi2)
      );

      return {
        lat: toDeg(phi2),
        lon: ((toDeg(lambda2) + 540) % 360) - 180
      };
    }

    function getCardinalDirection(deg) {
      const dirs = ['N', 'NNE', 'NE', 'ENE', 'E', 'ESE', 'SE', 'SSE', 'S', 'SSO', 'SO', 'OSO', 'O', 'ONO', 'NO', 'NNO'];
      const index = Math.round(((deg % 360) / 22.5)) % 16;
      return dirs[index];
    }

    function toRad(deg) {
      return (deg * Math.PI) / 180;
    }

    function toDeg(rad) {
      return (rad * 180) / Math.PI;
    }
